# 后端服务埋点方案与管理层看板设计

## 一、埋点方案设计

### 1.1 核心指标体系

#### 业务指标分层（KPI树）

```
第一层：战略级指标（管理层核心关注）
├── GMV（交易总额）
├── DAU/MAU（活跃用户）
├── 订单量
├── 客单价
└── 转化率

第二层：业务过程指标（运营层关注）
├── 用户行为：注册、登录、浏览、下单、支付
├── 流量来源：渠道转化、ROI
├── 商品指标：SKU动销率、库存周转
└── 服务质量：投诉率、NPS

第三层：系统性能指标（技术层关注）
├── API响应时间
├── 错误率
├── 服务可用性
└── 资源使用率
```

#### 管理层核心看板指标

| 指标名称 | 计算方式 | 更新频率 | 目标值 |
|---------|---------|---------|--------|
| 日活用户(DAU) | 当日去重登录用户数 | 实时 | 100,000+ |
| 订单转化率 | 下单用户/访问用户 | 实时 | 5%+ |
| 客单价 | GMV/订单数 | 日 | 500元+ |
| 支付成功率 | 支付成功订单/总订单 | 实时 | 98%+ |
| 30天留存率 | 第30天回访用户/首日用户 | 周 | 30%+ |

### 1.3 埋点实施方式

#### 方式一：代码埋点（推荐）

```python
# 示例：Python后端埋点SDK
import logging
from datetime import datetime
from typing import Dict, Any
import json

class EventTracker:
    """事件追踪器"""

    def __init__(self, kafka_producer, topic: str):
        self.producer = kafka_producer
        self.topic = topic
        self.logger = logging.getLogger(__name__)

    def track_event(self, event_type: str, event_name: str,
                    user_id: str, properties: Dict[str, Any] = None):
        """记录业务事件"""

        event = {
            "event_id": self._generate_event_id(),
            "user_id": user_id,
            "event_type": event_type,
            "event_name": event_name,
            "event_time": datetime.now().isoformat(),
            "event_date": datetime.now().strftime("%Y-%m-%d"),
            "properties": properties or {},
            "created_at": datetime.now().isoformat()
        }

# 发送到Kafka
        try:
            self.producer.send(self.topic, json.dumps(event).encode('utf-8'))
            self.logger.info(f"Event tracked: {event_name}")
        except Exception as e:
            self.logger.error(f"Failed to track event: {e}")

    def track_order_created(self, user_id: str, order_id: str,
                            amount: float, category: str, channel: str):
        """记录订单创建事件"""

        properties = {
            "order_id": order_id,
            "amount": amount,
            "category": category,
            "channel": channel,
            "timestamp": datetime.now().isoformat()
        }

        self.track_event(
            event_type="business",
            event_name="order_created",
            user_id=user_id,
            properties=properties
        )

    def track_order_paid(self, user_id: str, order_id: str,
                         payment_amount: float, payment_method: str):
        """记录订单支付事件"""

        properties = {
            "order_id": order_id,
            "payment_amount": payment_amount,
            "payment_method": payment_method,
            "payment_time": datetime.now().isoformat()
        }

        self.track_event(
            event_type="business",
            event_name="order_paid",
            user_id=user_id,
            properties=properties
        )

    def track_api_call(self, api_path: str, method: str,
                       response_time_ms: int, status_code: int,
                       user_id: str = None):
        """记录API调用事件"""

        properties = {
            "api_path": api_path,
            "method": method,
            "response_time_ms": response_time_ms,
            "status_code": status_code,
            "is_success": 200 <= status_code < 300
        }

        self.track_event(
            event_type="system",
            event_name="api_call",
            user_id=user_id or "system",
            properties=properties
        )
```

#### 方式二：AOP切面埋点（Java示例）

```java
// 注解定义
@Target(ElementType.METHOD)
@Retention(RetentionPolicy.RUNTIME)
public @interface TrackEvent {
    String eventType() default "business";
    String eventName();
    String description() default "";
}

// 切面实现
@Aspect
@Component
public class EventTrackingAspect {

    @Autowired
    private EventTracker eventTracker;

    @Around("@annotation(trackEvent)")
    public Object trackEvent(ProceedingJoinPoint joinPoint,
                             TrackEvent trackEvent) throws Throwable {
        // 获取用户ID
        String userId = SecurityContextHolder.getContext()
                          .getAuthentication().getName();

        Map<String, Object> properties = new HashMap<>();
        properties.put("method", joinPoint.getSignature().getName());
        properties.put("args", joinPoint.getArgs());

        try {
            Object result = joinPoint.proceed();

            // 记录成功事件
            eventTracker.trackEvent(
                trackEvent.eventType(),
                trackEvent.eventName() + "_success",
                userId,
                properties
            );

            return result;
        } catch (Exception e) {
            // 记录失败事件
            properties.put("error", e.getMessage());
            eventTracker.trackEvent(
                trackEvent.eventType(),
                trackEvent.eventName() + "_failed",
                userId,
                properties
            );
            throw e;
        }
    }
}

// 使用示例
@Service
public class OrderService {

    @TrackEvent(eventType = "business", eventName = "order_create",
                description = "创建订单")
    public Order createOrder(CreateOrderRequest request) {
        // 业务逻辑
        return order;
    }

    @TrackEvent(eventType = "business", eventName = "order_pay",
                description = "支付订单")
    public PaymentResult payOrder(String orderId, PayRequest request) {
        // 支付逻辑
        return paymentResult;
    }
```

```sql
-- ClickHouse建表语句
-- 1. 实时用户统计物化视图
CREATE MATERIALIZED VIEW mv_realtime_user_stats
ENGINE = AggregatingMergeTree()
PARTITION BY toYYYYMMDD(event_time)
ORDER BY (event_date, toStartOfHour(event_time))
AS SELECT
    event_date,
    toStartOfHour(event_time) as hour,
    uniqState(user_id) as unique_users,
    countState() as total_events
FROM fact_user_events
GROUP BY event_date, hour;

-- 2. 订单统计物化视图
CREATE MATERIALIZED VIEW mv_order_stats
ENGINE = AggregatingMergeTree()
PARTITION BY toYYYYMMDD(event_time)
ORDER BY (event_date, toStartOfHour(event_time))
AS SELECT
    event_date,
    toStartOfHour(event_time) as hour,
    countState() as order_count,
    sumState(order_amount) as total_amount,
    uniqState(user_id) as unique_buyers
FROM fact_order_events
WHERE event_type = 'create'
GROUP BY event_date, hour;

-- 查询示例：获取今日实时GMV
SELECT
    event_date,
    hour,
    uniqMerge(unique_users) as users,
    sumMerge(total_amount) as gmv,
    countMerge(order_count) as orders
FROM mv_order_stats
WHERE event_date = today()
GROUP BY event_date, hour
ORDER BY hour DESC
LIMIT 1;
```

### 2.4 数据查询优化

```sql
-- 常用查询优化示例

-- 1. 实时DAU查询（使用物化视图）
SELECT
    uniqMerge(unique_users) as dau
FROM mv_realtime_user_stats
WHERE event_date = today();

-- 2. 渠道ROI分析查询
WITH channel_stats AS (
    SELECT
        channel_source,
        count(DISTINCT user_id) as users,
        sum(order_amount) as gmv
    FROM fact_order_events
    WHERE event_type = 'pay'
      AND event_date >= today() - 30
    GROUP BY channel_source
),
channel_cost AS (
    SELECT
        channel,
        sum(cost) as total_cost
    FROM fact_marketing_cost
    WHERE date >= today() - 30
    GROUP BY channel
)
SELECT
    s.channel_source as channel,
    s.users,
    s.gmv,
    c.total_cost,
    s.gmv / c.total_cost as roi,
    c.total_cost / s.users as cac
FROM channel_stats s
LEFT JOIN channel_cost c ON s.channel_source = c.channel
ORDER BY roi DESC;

-- 3. 用户留存分析（30天留存）
WITH day0_users AS (
    SELECT DISTINCT user_id
    FROM fact_user_events
    WHERE event_date = today() - 30
),
day30_users AS (
    SELECT DISTINCT user_id
    FROM fact_user_events
    WHERE event_date = today()
)
SELECT
    count(DISTINCT d30.user_id) as retained_users,
    count(DISTINCT d0.user_id) as total_users,
    count(DISTINCT d30.user_id) / count(DISTINCT d0.user_id) * 100 as retention_rate
FROM day0_users d0
LEFT JOIN day30_users d30 ON d0.user_id = d30.user_id;

-- 4. 创建索引优化查询性能
CREATE INDEX idx_user_events_date_user
ON fact_user_events (event_date, user_id);

CREATE INDEX idx_order_events_date_type
ON fact_order_events (event_date, event_type);

CREATE INDEX idx_api_calls_path_time
ON fact_api_calls (api_path, request_time);
```
